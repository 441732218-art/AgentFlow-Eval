import os
import re
import ast
from pathlib import Path

# ==================== CONFIGURATION (Modify Here) ====================
ROOT_DIR = r"D:\AgentFlow-Eval"                         # Your project root
OUTPUT_DIR = r"D:\AgentFlow-Eval\copyright_output"     # Output directory
SOFTWARE_NAME = "AgentFlow-Eval 源程序鉴别材料"            # Software title
VERSION = "V1.0.0"                                      # Version string
AUTHOR = "李凯昕"                                 # Copyright Owner
TOTAL_TARGET_LINES = 3000                               # Standard soft-copyright requirement (60 pages * 50 lines)
LINES_PER_PAGE = 50
# =====================================================================

os.makedirs(OUTPUT_DIR, exist_ok=True)

# File patterns to exclude (Generic / Non-original / Auto-generated)
EXCLUDE_PATTERNS = [
    r"__init__\.py$",
    r"site-packages",
    r"venv",
    r"\.venv",
    r"migrations",
    r"test",
    r"setup\.py",
    r"config\.py",
]

def is_excluded(file_path: Path) -> bool:
    path_str = str(file_path).lower()
    return any(re.search(pattern, path_str, re.IGNORECASE) for pattern in EXCLUDE_PATTERNS)

def calculate_originality_score(file_path: Path) -> float:
    """
    Ranks files based on business logic complexity to emphasize originality.
    Higher scores mean deeper algorithmic/business logic.
    """
    try:
        content = file_path.read_text(encoding="utf-8", errors="ignore")
        tree = ast.parse(content)
        
        score = 0.0
        for node in ast.walk(tree):
            if isinstance(node, ast.FunctionDef) or isinstance(node, ast.AsyncFunctionDef):
                score += 3.0  # Functions/Methods
            elif isinstance(node, ast.ClassDef):
                score += 5.0  # Custom Data/Logic Classes
            elif isinstance(node, (ast.If, ast.For, ast.While, ast.Try)):
                score += 1.5  # Control Flows
            elif isinstance(node, ast.Call):
                score += 0.5  # Invocations
        
        # Penalize short or excessively trivial files
        lines = [line.strip() for line in content.splitlines() if line.strip() and not line.strip().startswith('#')]
        if len(lines) < 15:
            score *= 0.2
            
        return score
    except Exception:
        return 0.0

def clean_and_collect_lines(file_path: Path):
    """Reads file and returns non-empty, valid lines with a soft header."""
    lines = []
    rel_path = file_path.relative_to(ROOT_DIR)
    lines.append(f"# File: {rel_path}")
    lines.append(f"# (c) 2026 {SOFTWARE_NAME} Author: {AUTHOR}")
    
    try:
        raw_lines = file_path.read_text(encoding="utf-8", errors="ignore").splitlines()
        for line in raw_lines:
            stripped = line.strip()
            # Retain code, ignore empty lines and standalone comment lines for maximum density
            if stripped and not stripped.startswith("#"):
                lines.append(line)
    except Exception as e:
        print(f"[WARN] Failed to read {file_path}: {e}")
        
    return lines

def main():
    print(f"[1/4] Scanning files for high-originality code in: {ROOT_DIR}")
    py_files = [p for p in Path(ROOT_DIR).rglob("*.py") if not is_excluded(p)]
    
    # Rank files by AST Originality Score
    scored_files = []
    for file_path in py_files:
        score = calculate_originality_score(file_path)
        if score > 0:
            scored_files.append((score, file_path))
            
    # Sort files descending by originality
    scored_files.sort(key=lambda x: x[0], reverse=True)
    print(f"[INFO] Found {len(scored_files)} valid logic files ranked by originality score.")

    # Pool all high-value lines
    all_lines = []
    for score, file_path in scored_files:
        lines = clean_and_collect_lines(file_path)
        all_lines.extend(lines)

    total_lines_count = len(all_lines)
    print(f"[INFO] Total clean effective lines extracted: {total_lines_count}")

    # Trim/Sample exact 3000 lines (First 1500 + Last 1500) according to national software copyright rules
    if total_lines_count >= TOTAL_TARGET_LINES:
        half_target = TOTAL_TARGET_LINES // 2
        head_lines = all_lines[:half_target]
        tail_lines = all_lines[-half_target:]
        
        # Insert explicit omission indicator as required by copyright audit standards
        omission_notice = [
            "# " + "=" * 60,
            "# [中央软件著作权审查标记] 此处省略中间部分源代码 (Source Code Omitted Here)",
            "# " + "=" * 60
        ]
        final_lines = head_lines + omission_notice + tail_lines
    else:
        print("[WARN] Codebase contains less than 3,000 effective lines. Including all available lines.")
        final_lines = all_lines

    # Output text preview file
    txt_out = Path(OUTPUT_DIR) / "source_code_raw.txt"
    txt_out.write_text("\n".join(final_lines), encoding="utf-8")
    print(f"[2/4] Raw filtered code saved -> {txt_out}")

    # Build HTML for PDF Export (Exact 60 pages with running headers & footers)
    print("[3/4] Formatting standard HTML layout for copyright submission...")
    
    pages = []
    # Force EXACT 3000 lines slicing for HTML generator (3000 lines / 50 per page = 60 pages)
    trimmed_code_lines = [l for l in final_lines if not l.startswith("# ==")]
    if len(trimmed_code_lines) > TOTAL_TARGET_LINES:
        trimmed_code_lines = trimmed_code_lines[:1500] + trimmed_code_lines[-1500:]

    total_pages = (len(trimmed_code_lines) + LINES_PER_PAGE - 1) // LINES_PER_PAGE

    html_content = f"""<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8">
    <title>{SOFTWARE_NAME} {VERSION} 源程序鉴别材料</title>
    <style>
        @page {{
            size: A4 portrait;
            margin: 20mm 15mm 20mm 15mm;
        }}
        body {{
            font-family: "Courier New", Consolas, monospace;
            font-size: 9.5pt;
            line-height: 1.35;
            color: #000;
            margin: 0;
            padding: 0;
        }}
        .page {{
            page-break-after: always;
            height: 245mm;
            position: relative;
            box-sizing: border-box;
        }}
        .header {{
            border-bottom: 1px solid #000;
            padding-bottom: 3px;
            margin-bottom: 10px;
            display: flex;
            justify-content: space-between;
            font-weight: bold;
            font-size: 10pt;
        }}
        .footer {{
            position: absolute;
            bottom: 0;
            left: 0;
            right: 0;
            text-align: center;
            border-top: 1px solid #ccc;
            padding-top: 3px;
            font-size: 9pt;
        }}
        pre {{
            margin: 0;
            white-space: pre-wrap;
            word-wrap: break-word;
            font-family: inherit;
        }}
    </style>
</head>
<body>
"""

    for page_idx in range(total_pages):
        page_num = page_idx + 1
        page_code_chunk = trimmed_code_lines[page_idx * LINES_PER_PAGE : (page_idx + 1) * LINES_PER_PAGE]
        
        # Check if we are at the center boundary (Page 30/31) to show omission notice
        code_text = "\n".join(page_code_chunk)
        
        html_content += f"""
        <div class="page">
            <div class="header">
                <span>{SOFTWARE_NAME} {VERSION}</span>
                <span>著作权人：{AUTHOR}</span>
            </div>
            <pre>{code_text}</pre>
            <div class="footer">
                第 {page_num} 页 / 共 {total_pages} 页
            </div>
        </div>
        """

    html_content += "</body></html>"

    html_out = Path(OUTPUT_DIR) / "source_code_60pages.html"
    html_out.write_text(html_content, encoding="utf-8")
    print(f"[4/4] Complete! HTML generated -> {html_out}")
    print("\n" + "="*60)
    print("【导出步骤】:")

    print("3. 目标打印机选择: '另存为 PDF'")
    print("4. 布局选择: '纵向', 勾选 '背景图形'")
    print("5. 导出并确认总页数为 60 页，页眉页码清晰。")
    print("="*60)

if __name__ == "__main__":
    main()

   393	    """Probe database and Redis; return per-service status strings."""

   411	            skip_redis = True

   413	            skip_redis = True
   414	    except Exception:
   415	        pass
   416	
   417	    if skip_redis:
   418	        services["redis"] = "skipped (lite/eager)"
